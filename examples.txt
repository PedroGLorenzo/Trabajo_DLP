letrec add : Int -> Int -> Int =
    lambda i1: Int.
    lambda i2: Int.
        case i1 of
        <pos=p1> =>
            case i2 of
            <pos=p2> => 
                if iszero p2 then
                    i1
                else
                    add <pos=succ p1> as Int <pos=pred p2> as Int
            | <zero=z> => 
                if iszero p1 then
                    <zero=true> as Int
                else
                    <pos=p1> as Int
            | <neg=n2> => 
                if iszero n2 then
                    i1
                else if iszero (pred p1) then
                    add <zero=true> as Int <neg=pred n2> as Int
                else
                    add <pos=pred p1> as Int <neg=pred n2> as Int
        | <zero=z> =>
           case i2 of
            <pos=p2> => i2
            |<zero=z> => i2
            |<neg=n2> => i2
        | <neg=n1> =>
            case i2 of
            <pos=p2> => 
                if iszero p2 then
                    i1
                else if iszero (pred n1) then
                    add <zero=true> as Int <pos=pred p2> as Int    
                else
                    add <neg=pred n1> as Int <pos=pred p2> as Int
            | <zero=z> => 
                if iszero n1 then
                    <zero=true> as Int
                else
                    i2
            | <neg=n2> => 
                if iszero n2 then
                    i2
                else
